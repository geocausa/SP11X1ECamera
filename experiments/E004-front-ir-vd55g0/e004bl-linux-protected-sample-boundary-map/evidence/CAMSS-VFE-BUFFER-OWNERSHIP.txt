   578	 * vfe_queue_buffer_v2 - Add empty buffer
   579	 * @vid: Video device structure
   580	 * @buf: Buffer to be enqueued
   581	 *
   582	 * Add an empty buffer - depending on the current number of buffers it will be
   583	 * put in pending buffer queue or directly given to the hardware to be filled.

   585	 * Return 0 on success or a negative error code otherwise

   587	int vfe_queue_buffer_v2(struct camss_video *vid,
   588				struct camss_buffer *buf)
   589	{
   590		struct vfe_line *line = container_of(vid, struct vfe_line, video_out);
   591		struct vfe_device *vfe = to_vfe(line);
   592		const struct vfe_hw_ops *ops = vfe->res->hw_ops;
   593		struct vfe_output *output;
   594		unsigned long flags;
   595	
   596		output = &line->output;
   597	
   598		spin_lock_irqsave(&vfe->output_lock, flags);
   599	
   600		if (output->state == VFE_OUTPUT_ON &&
   601		    output->gen2.active_num < 2) {
   602			output->buf[output->gen2.active_num++] = buf;
   603			ops->vfe_wm_update(vfe, output->wm_idx[0],
   604					   buf->addr[0], line);
   605			ops->reg_update(vfe, line->id);
   606		} else {
   607			vfe_buf_add_pending(output, buf);
   608		}
   609	
   610		spin_unlock_irqrestore(&vfe->output_lock, flags);
   611	
   612		return 0;
   613	}
