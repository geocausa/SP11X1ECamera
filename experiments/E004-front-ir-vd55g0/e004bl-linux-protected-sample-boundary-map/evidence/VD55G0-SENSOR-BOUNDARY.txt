   956	static int vd55g0_stream_enable(struct vd55g0_dev *sensor)
   957	{
   958		struct i2c_client *client = v4l2_get_subdevdata(&sensor->sd);
   959		int ret;
   960	
   961		ret = pm_runtime_get_sync(&client->dev);
   962		if (ret < 0) {
   963			pm_runtime_put_autosuspend(&client->dev);
   964			return ret;
   965		}
   966	
   967		/* pm_runtime_get_sync() can return 1 as a valid return code */
   968		ret = 0;
   969	
   970		ret = vd55g0_set_gpios(sensor);
   971		if (ret)
   972			goto err_rpm_put;
   973	
   974		vd55g0_write_reg(sensor, VD55G0_REG_FORMAT_CTRL,
   975				 get_bpp_by_code(sensor->fmt.code), &ret);
   976		vd55g0_write_reg(sensor, VD55G0_REG_OIF_IMG_CTRL,
   977				 get_data_type_by_code(sensor->fmt.code), &ret);
   978		if (ret)
   979			goto err_rpm_put;
   980	
   981		ret = vd55g0_apply_frame_format(sensor);
   982		if (ret)
   983			goto err_rpm_put;
   984	
   985		ret = vd55g0_apply_settings(sensor);
   986		if (ret)
   987			goto err_rpm_put;
   988	
   989		ret = vd55g0_write_reg(sensor, VD55G0_REG_SW_STBY,
   990				       VD55G0_SW_STBY_START_STREAM, NULL);
   991		if (ret)
   992			goto err_rpm_put;
   993	
   994		ret = vd55g0_poll_reg(sensor, VD55G0_REG_SW_STBY, 0, VD55G0_TIMEOUT_MS);
   995		if (ret)
   996			goto err_rpm_put;
   997	
   998		ret = vd55g0_wait_state(sensor, VD55G0_SYSTEM_FSM_STREAMING,
   999					VD55G0_TIMEOUT_MS);
  1000		if (ret)
  1001			goto err_rpm_put;
  1002	
  1003		return 0;
  1004	
  1005	err_rpm_put:
  1006		pm_runtime_put(&client->dev);
  1007		return ret;
  1008	}
